70# region Storage header 

71# The `storage` header names which of the two consumers of the online topic is meant to 

72# ingest a record, mirroring the `storage` argument of 

73# [`FeatureGroup.insert`][hsfs.feature_group.FeatureGroup.insert]: 

74# 

75# absent both OnlineFS and the offline materialization job ingest the record 

76# b"online" OnlineFS only; the materialization job skips the record 

77# b"offline" the materialization job only; OnlineFS skips the record 

78# 

79# Older clients wrote a per-row flag into the same header instead: b"0" meant "skip 

80# online" (equivalent to b"offline") and b"1" meant "ingest online" while the record was 

81# still materialized offline (equivalent to absent). 

82# Both consumers keep reading b"0" as b"offline"; the producers here no longer emit either 

83# value. 

84# Only set the header when the record has a single destination: leaving it out is both the 

85# default and the cheapest option on the wire, which matters per record at ingestion scale. 

86_STORAGE_ONLINE = "online" 

87_STORAGE_OFFLINE = "offline" 

88# endregion

166def _online_delete_fill_values( 

167 feature_group: FeatureGroup | ExternalFeatureGroup, 

168) -> dict[str, Any]: 

169 """Null fill values for every non-primary-key field in the feature group schema. 

170 

171 An online delete tombstone is serialized against the full feature group Avro 

172 schema, but the caller only needs to pass the primary key (the offline delete 

173 contract). 

174 Non-key fields are filled with null so the record serializes; OnlineFS deletes 

175 by primary key and discards the values. 

176 Online feature group schemas make every non-key field a nullable union, so null 

177 always serializes. 

178 """ 

179 primary_key = set(feature_group.primary_key) 

180 fields = json.loads(feature_group.avro_schema)["fields"] 

181 return {field["name"]: None for field in fields if field["name"] not in primary_key}

184def _get_headers( 

185 feature_group: FeatureGroup | ExternalFeatureGroup, 

186 num_entries: int | None = None, 

187 options: dict[str, Any] | None = None, 

188 operation: str | None = None, 

189 storage: str | None = None, 

190) -> dict[str, bytes]: 

191 """Kafka headers for the records of one write to the online topic. 

192 

193 `storage` is the destination of this write: `"online"` for records only OnlineFS 

194 should ingest, `"offline"` for records only the offline materialization job should 

195 ingest, `None` when both consume them. 

196 See the storage header contract at the top of this module. 

197 """ 

198 # custom headers for hopsworks onlineFS 

199 headers = { 

200 "projectId": str(feature_group.feature_store.project_id).encode("utf8"), 

201 "featureGroupId": str(feature_group._id).encode("utf8"), 

202 "subjectId": str(feature_group.subject["id"]).encode("utf8"), 

203 } 

204 

205 # operation header tells OnlineFS whether the message is an upsert (absent) or a 

206 # delete tombstone ("delete"); OnlineFS deletes the row by primary key on "delete". 

207 if operation is not None: 

208 headers["operation"] = operation.encode("utf8") 

209 

210 if storage is not None: 

211 headers["storage"] = storage.encode("utf8") 

212 

213 online_ingestion_options = ( 

214 options.get("online_ingestion_options") if options else None 

215 ) 

216 if online_ingestion_options and online_ingestion_options.get("upsert_if_newer"): 

217 headers["upsertIfNewer"] = b"1" 

218 

219 # An offline-only write reaches no online store, so it gets no online ingestion to 

220 # report progress against: creating one would leave it forever short of its entries. 

221 if feature_group.online_enabled and storage != _STORAGE_OFFLINE: 

222 # setup online ingestion id 

223 online_ingestion_instance = ( 

224 online_ingestion_api.OnlineIngestionApi()._create_online_ingestion( 

225 feature_group, online_ingestion.OnlineIngestion(num_entries=num_entries) 

226 ) 

227 ) 

228 headers["onlineIngestionId"] = str(online_ingestion_instance.id).encode("utf8") 

229 

230 return headers

233def _wait_for_online_ingestion( 

234 feature_group: FeatureGroup | ExternalFeatureGroup, 

235 headers: dict[str, bytes], 

236 options: dict[str, Any], 

237) -> None: 

238 """Block until OnlineFS has ingested the records written with `headers`, if `options` asks to wait. 

239 

240 Returns at once for records that carry no online ingestion id, such as those of an offline-only write. 

241 Warns and returns when the ingestion no longer exists, as the backend prunes the oldest ingestions of a feature group. 

242 """ 

243 if not options.get("wait_for_online_ingestion", False): 

244 return 

245 online_ingestion_id = headers.get("onlineIngestionId") 

246 if online_ingestion_id is None: 

247 return 

248 online_ingestion_id = int(online_ingestion_id) 

249 # Not get_latest_online_ingestion: the backend picks the latest by id, and NDB hands out 

250 # auto-increment ids in per-mysqld blocks, so with several mysqlds the latest by id can be 

251 # an older ingestion that has already completed. 

252 online_ingestion_instance = feature_group.get_online_ingestion(online_ingestion_id) 

253 if online_ingestion_instance is None: 

254 warnings.warn( 

255 f"Online ingestion {online_ingestion_id} of feature group '{feature_group.name}' " 

256 "was pruned by the backend before the write could wait for it, so the write returns " 

257 "without knowing whether its rows have reached the online feature store.", 

258 stacklevel=1, 

259 ) 

260 return 

261 online_ingestion_instance.wait_for_completion( 

262 options=options.get("online_ingestion_options", {}) 

263 )

275def _get_watermark_offsets( 

276 consumer: Consumer, partition: TopicPartition, timeout: float 

277) -> tuple[int, int]: 

278 """Read a partition's watermarks, waiting out a partition whose leader is not serving yet. 

279 

280 A topic is listed as soon as it is created, but each broker answers for its partitions 

281 only once it has loaded them, a fraction of a second later; a partition whose leader 

282 just moved is the same. Until then the broker refuses the read with one of the errors 

283 below, so the read is retried with a short backoff for up to `timeout` seconds rather 

284 than failing the insert that asked for the offsets. 

285 """ 

286 deadline = time.monotonic() + timeout 

287 delay = 0.1 

288 while True: 

289 try: 

290 return consumer.get_watermark_offsets(partition) 

291 except KafkaException as e: 

292 error = e.args[0] if e.args else None 

293 transient = isinstance(error, KafkaError) and error.code() in ( 

294 KafkaError.NOT_LEADER_FOR_PARTITION, 

295 KafkaError.LEADER_NOT_AVAILABLE, 

296 KafkaError.UNKNOWN_TOPIC_OR_PART, 

297 ) 

298 if not transient or time.monotonic() + delay > deadline: 

299 raise 

300 time.sleep(delay) 

301 delay = min(delay * 2, 1.0)

304@_uses_confluent_kafka 

305def _kafka_get_offsets( 

306 topic_name: str, 

307 feature_store_id: int, 

308 offline_write_options: dict[str, Any], 

309 high: bool, 

310) -> str: 

311 consumer = _init_kafka_consumer(feature_store_id, offline_write_options) 

312 try: 

313 timeout = offline_write_options.get("kafka_timeout", 6) 

314 topics = consumer.list_topics(timeout=timeout).topics 

315 if topic_name not in topics: 

316 return "" 

317 offsets = "" 

318 tuple_value = int(high) 

319 for partition_metadata in topics.get(topic_name).partitions.values(): 

320 partition = TopicPartition( 

321 topic=topic_name, partition=partition_metadata.id 

322 ) 

323 watermarks = _get_watermark_offsets(consumer, partition, timeout) 

324 offsets += f",{partition_metadata.id}:{watermarks[tuple_value]}" 

325 return f"{topic_name + offsets}" 

326 finally: 

327 consumer.close()

330@_uses_confluent_kafka 

331def _kafka_get_offsets_for_times( 

332 topic_name: str, 

333 feature_store_id: int, 

334 offline_write_options: dict[str, Any], 

335 timestamp: int, 

336) -> str: 

337 """Look up the offsets a topic has to be read from to skip everything older than a timestamp. 

338 

339 Returns the `topic,partition:offset,...` string 

340 [`_kafka_get_offsets`][hsfs.core.kafka_engine._kafka_get_offsets] returns, holding for 

341 each partition the earliest offset whose record is at or after `timestamp`. 

342 A partition whose records all predate `timestamp` contributes the high watermark it had 

343 before the lookup, since it held nothing worth reading then; a record appended after the 

344 lookup lands at or past that watermark and is still read. 

345 A partition Kafka could not answer for contributes 

346 its low watermark, so an unanswered lookup reads too much rather than too little. 

347 The empty string is returned when the topic does not exist. 

348 

349 Parameters: 

350 topic_name: Name of the topic to look the offsets up in. 

351 feature_store_id: Id of the feature store the topic belongs to. 

352 offline_write_options: Options the consumer is built from, honouring `kafka_timeout`. 

353 timestamp: Unix timestamp in milliseconds to look the offsets up at. 

354 

355 Returns: 

356 The offsets, in the same format as `_kafka_get_offsets`. 

357 """ 

358 consumer = _init_kafka_consumer(feature_store_id, offline_write_options) 

359 try: 

360 timeout = offline_write_options.get("kafka_timeout", 6) 

361 topics = consumer.list_topics(timeout=timeout).topics 

362 if topic_name not in topics: 

363 return "" 

364 

365 partitions = [ 

366 partition_metadata.id 

367 for partition_metadata in topics.get(topic_name).partitions.values() 

368 ] 

369 # Captured before the lookup: a high watermark read after it could already be past 

370 # a record appended in between, which the lookup did not see and the read would 

371 # then skip. 

372 highs = { 

373 partition: _get_watermark_offsets( 

374 consumer, TopicPartition(topic=topic_name, partition=partition), timeout 

375 )[1] 

376 for partition in partitions 

377 } 

378 lookups = [ 

379 TopicPartition(topic=topic_name, partition=partition, offset=timestamp) 

380 for partition in partitions 

381 ] 

382 offsets = "" 

383 for result in consumer.offsets_for_times(lookups, timeout=timeout): 

384 # Read after the lookup, since retention can drop the record any offset below 

385 # points at. 

386 low, _ = _get_watermark_offsets( 

387 consumer, 

388 TopicPartition(topic=topic_name, partition=result.partition), 

389 timeout, 

390 ) 

391 if result.error is not None: 

392 # Reading from the low watermark is what this function is meant to avoid, 

393 # but a partition Kafka would not answer for has no offset to trust, and 

394 # re-reading records is recoverable where skipping them is not. 

395 offset = low 

396 elif result.offset < 0: 

397 # Kafka answers a timestamp past the last record with a negative offset: 

398 # every record the partition holds is older than the timestamp, so the 

399 # reader belongs at the end of it. 

400 offset = max(highs[result.partition], low) 

401 else: 

402 offset = max(result.offset, low) 

403 offsets += f",{result.partition}:{offset}" 

404 

405 return f"{topic_name + offsets}" 

406 finally: 

407 consumer.close()

410def _kafka_produce( 

411 producer: Producer, 

412 key: str, 

413 encoded_row: bytes, 

414 topic_name: str, 

415 headers: dict[str, bytes], 

416 acked: callable, 

417 debug_kafka: bool = False, 

418) -> None: 

419 while True: 

420 # if BufferError is thrown, we can be sure, message hasn't been send so we retry 

421 try: 

422 # produce 

423 producer.produce( 

424 topic=topic_name, 

425 key=key, 

426 value=encoded_row, 

427 callback=acked, 

428 headers=headers, 

429 ) 

430 

431 # Trigger internal callbacks to empty op queue 

432 producer.poll(0) 

433 break 

434 except BufferError as e: 

435 if debug_kafka: 

436 print(f"Caught: {e}") 

437 # backoff for 1 second 

438 producer.poll(1)

451def _get_encoder_func(writer_schema: str) -> callable: 

452 if HAS_FAST_AVRO: 

453 schema = json.loads(writer_schema) 

454 parsed_schema = parse_schema(schema) 

455 return lambda record, outf: schemaless_writer(outf, parsed_schema, record) 

456 

457 if not HAS_AVRO: 

458 raise ModuleNotFoundError(avro_not_installed_message) 

459 

460 parsed_schema = avro.schema.parse(writer_schema) 

461 writer = avro.io.DatumWriter(parsed_schema) 

462 return lambda record, outf: writer.write(record, avro.io.BinaryEncoder(outf))

465def _encode_row(complex_feature_writers, writer, row): 

466 # transform special data types 

467 # here we might need to handle also timestamps and other complex types 

468 # possible optimizaiton: make it based on type so we don't need to loop over 

469 # all keys in the row 

470 if isinstance(row, dict): 

471 for k in row: 

472 # NaT is a datetime subclass, so it would pass the timezone branch below and reach avro as a 

473 # timestamp it cannot write. 

474 if HAS_PANDAS and row[k] is pd.NaT: 

475 row[k] = None 

476 continue 

477 # for avro to be able to serialize them, they need to be python data types 

478 if HAS_NUMPY and isinstance(row[k], np.ndarray): 

479 row[k] = row[k].tolist() 

480 if HAS_PANDAS and isinstance(row[k], pd.Timestamp): 

481 row[k] = row[k].to_pydatetime() 

482 if isinstance(row[k], datetime) and row[k].tzinfo is None: 

483 row[k] = row[k].replace(tzinfo=timezone.utc) 

484 if HAS_PANDAS and isinstance(row[k], pd._libs.missing.NAType): 

485 row[k] = None 

486 # encode complex features 

487 row = _encode_complex_features(complex_feature_writers, row) 

488 # encode feature row 

489 with BytesIO() as outf: 

490 writer(row, outf) 

491 return outf.getvalue()

553def _delivery_error(err: KafkaError, msg: Any) -> FeatureStoreException: 

554 """Build the exception raised when Kafka fails to deliver a produced row. 

555 

556 Any delivery error means the row never reached the topic, so the online feature store will 

557 never ingest it. 

558 Reporting it is what keeps a partially delivered insert from looking like a successful one. 

559 

560 `confluent_kafka.KafkaError` is an odd type to raise: CPython accepts it as the operand of 

561 `raise`, yet it does not subclass `BaseException`, so `except Exception` never matches it and 

562 only an explicit `except KafkaError` would. 

563 Raising it directly would therefore slip past a caller's error handling, so it is wrapped in a 

564 `KafkaException` cause, which is an ordinary exception and keeps the original error code 

565 reachable for callers that need to branch on it. 

566 """ 

567 location = "" 

568 # The message handle may be unusable; the error itself is still worth reporting. 

569 with contextlib.suppress(AttributeError, TypeError): 

570 location = f" to topic '{msg.topic()}' partition {msg.partition()}" 

571 

572 hint = "" 

573 if err.code() in (KafkaError.MSG_SIZE_TOO_LARGE, KafkaError.RECORD_LIST_TOO_LARGE): 

574 hint = ( 

575 " The row exceeds the maximum message size accepted by the broker." 

576 " Reduce the size of the row's values, or raise the topic's 'max.message.bytes'" 

577 " together with the producer's 'message.max.bytes'" 

578 " (via the 'kafka_producer_config' write option)." 

579 ) 

580 

581 exception = FeatureStoreException( 

582 f"Failed to deliver row{location}: {err!s}.{hint}" 

583 " The insert is incomplete - this row was not written to the online feature store." 

584 ) 

585 exception.__cause__ = KafkaException(err) 

586 return exception
