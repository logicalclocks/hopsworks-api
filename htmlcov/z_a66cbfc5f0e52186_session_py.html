1"""``hops session`` — move a coding agent session between this machine and a pod. 

2 

3The mental model is a hand-off. ``push`` hands the session you are in to a 

4Hopsworks terminal pod (Claude Code is pre-installed in that image); the pod 

5lands it as soon as its browser Terminal tab is open, and from then on you can 

6close the laptop and keep driving it from that tab. ``pull`` reclaims it back onto this machine. ``new`` 

7starts a fresh session straight on the pod, ``list`` shows where things are, 

8``stop`` kills the terminal pod, and ``mirror`` streams the live pod terminal 

9to your laptop. 

10 

11The transport is symmetric. Each session is a JSONL that Claude Code stores at 

12``~/.claude/projects/<cwd-slug>/<session-id>.jsonl``; push uploads it into the 

13user's private HopsFS home under ``Users/<username>/teleport/<slug>/`` and pull 

14downloads it back. That home is mode 0700, so a session transcript is readable 

15only by its owner, not by other project members. Resume is scoped to the 

16working-directory slug, so both ends must run from a path that hashes to the 

17same slug for ``claude --resume`` to load. 

18 

19Every push/new also stages a ``<id>.teleport.json`` manifest. The pod carries a 

20landing hook that watches the teleport dataset and, when a manifest appears, 

21resumes the pushed session (or opens a new one) on its own and acks with a 

22``<id>.landed.json`` sidecar; push waits briefly for that ack and prints the 

23manual landing steps only when it does not arrive. 

24"""

133def _resolve_local_session(slug: str, session_id: str | None) -> Path: 

134 """Locate the local session JSONL to push for ``slug``. 

135 

136 Args: 

137 slug: The current directory's Claude Code slug. 

138 session_id: Explicit session id to push, or None to pick the 

139 most-recently-written session under this slug (the active one). 

140 

141 Returns: 

142 Path to the resolved ``<session-id>.jsonl``, or to its ``.jsonl.away`` 

143 twin when an earlier push already handed the session off, so that a 

144 repeated push ships the same content instead of failing. 

145 

146 Raises: 

147 click.ClickException: When no session directory or file is found, or 

148 the requested ``session_id`` does not exist here. 

149 """ 

150 proj_dir = _CLAUDE_PROJECTS / slug 

151 if not proj_dir.is_dir(): 

152 raise click.ClickException( 

153 f"No coding agent sessions found for this directory " 

154 f"(looked in {proj_dir}). Run this from the directory your " 

155 f"session is in." 

156 ) 

157 if session_id: 

158 for name in (f"{session_id}.jsonl", f"{session_id}.jsonl.away"): 

159 candidate = proj_dir / name 

160 if candidate.is_file(): 

161 return candidate 

162 raise click.ClickException(f"Session {session_id} not found under {proj_dir}.")

164 sessions = sorted( 

165 proj_dir.glob(pattern), key=lambda p: p.stat().st_mtime, reverse=True 

166 ) 

167 if sessions: 

168 return sessions[0] 

169 raise click.ClickException(f"No .jsonl sessions under {proj_dir}.")

229def _held_open(path: Path) -> bool | None: 

230 """Whether a live process holds ``path`` open, or None when undeterminable. 

231 

232 Asks ``lsof -t`` (pids only, quiet). None when ``lsof`` is absent or fails, 

233 so the caller can fall back to its mtime heuristic; an empty pid list means 

234 definitively "no process has it open". 

235 """ 

236 lsof = shutil.which("lsof") 

237 if not lsof: 

238 return None 

239 try: 

240 proc = subprocess.run( 

241 [lsof, "-t", "--", str(path)], 

242 capture_output=True, 

243 text=True, 

244 timeout=5, 

245 check=False, 

246 ) 

247 except (OSError, subprocess.SubprocessError): 

248 return None 

249 return bool(proc.stdout.strip())

252def _is_active_session(jsonl: Path) -> bool: 

253 """Heuristic for "the session you are currently in". 

254 

255 Active when it is the newest transcript written within the last two minutes, 

256 or when a live process still holds it open (``lsof``) regardless of recency. 

257 The open-fd check is what keeps push from renaming a ``claude`` session out 

258 from under the running process, newest or not; the mtime window covers hosts 

259 without ``lsof``. When active, the baton is recorded but the local copy is 

260 left in place. 

261 """ 

262 proj_dir = jsonl.parent 

263 newest = max( 

264 proj_dir.glob("*.jsonl"), key=lambda p: p.stat().st_mtime, default=None 

265 ) 

266 if newest == jsonl and (time.time() - jsonl.stat().st_mtime) < 120: 

267 return True 

268 return _held_open(jsonl) is True

314

344def _staged_lines_ahead( 

345 dataset_api, dest_dir: str, session_id: str, local_lines: list[str] 

346) -> int: 

347 """How many lines the staged transcript holds that ``local_lines`` does not. 

348 

349 Zero when nothing is staged, or when the staged copy is the local one or a 

350 prefix of it. Lines past the common prefix are work this machine never had 

351 (a pod's synced-back appends, a push from another machine), which a push 

352 would destroy. Nothing staged reads as zero only on a real 404; any other 

353 failure raises, since guessing here is what the check exists to prevent. 

354 """ 

355 remote = f"{dest_dir}/{session_id}.jsonl" 

356 with tempfile.TemporaryDirectory() as tmp: 

357 local = Path(tmp) / "staged.jsonl" 

358 try: 

359 dataset_api.download(remote, local_path=str(local), overwrite=True) 

360 except RestAPIError as exc: 

361 if _is_missing_path(exc): 

362 return 0 

363 raise click.ClickException( 

364 f"Could not read the staged session: {exc}" 

365 ) from exc 

366 except Exception as exc: # noqa: BLE001 

367 raise click.ClickException( 

368 f"Could not read the staged session: {exc}" 

369 ) from exc 

370 remote_lines = local.read_text(errors="ignore").splitlines() 

371 common = 0 

372 for a, b in zip(local_lines, remote_lines, strict=False): 

373 if a != b: 

374 break 

375 common += 1 

376 return len(remote_lines) - common

395def _transcript_relation(local: list[str], remote: list[str], baseline: int) -> str: 

396 """Classify how two append-only transcripts relate. 

397 

398 A Claude Code JSONL only ever grows (lines are appended, never rewritten), 

399 so the relationship is decided by the common line prefix. ``baseline`` is 

400 the line count recorded in the baton at the last hand-off. 

401 

402 Returns one of ``same``, ``fast_forward`` (local is a strict prefix of 

403 remote, take remote), ``local_ahead`` (remote is a strict prefix of local, 

404 keep local), ``baseline_mismatch`` (they already differ inside the handed-off 

405 prefix, so they are not one lineage), or ``diverged`` (shared prefix past the 

406 baseline, tails differ). 

407 """ 

408 common = 0 

409 for a, b in zip(local, remote, strict=False): 

410 if a != b: 

411 break 

412 common += 1 

413 if common == len(local) == len(remote): 

414 return "same" 

415 if common == len(local): 

416 return "fast_forward" 

417 if common == len(remote): 

418 return "local_ahead" 

419 if common < baseline: 

420 return "baseline_mismatch" 

421 return "diverged"

507def _upload_manifest( 

508 dataset_api, dest_dir: str, session_id: str, manifest: dict 

509) -> None: 

510 """Upload the teleport manifest — the last write of a push/new. 

511 

512 Not best-effort: the manifest is the pod watcher's trigger, so a silent drop 

513 would stage a session that never lands. A failure is surfaced loudly. 

514 

515 A prior land's ``.consumed`` marker and ``landed.json`` ack are deliberately 

516 NOT deleted here. Every marker is versioned by the manifest's ``pushed_at`` 

517 (the pod skips a manifest only when the marker records the same push, and 

518 ``_await_landing`` accepts only an ack echoing this push), so a re-push 

519 invalidates them without any delete. Deleting them out-of-band would also 

520 be a lost-write trap: the pod's FUSE mount caches directory entries, and a 

521 marker the pod writes into a name the CLI just deleted server-side can land 

522 in the stale unlinked inode and never become visible. 

523 """ 

524 with tempfile.TemporaryDirectory() as tmp: 

525 mpath = Path(tmp) / f"{session_id}.teleport.json" 

526 mpath.write_text(json.dumps(manifest)) 

527 try: 

528 dataset_api.upload( 

529 local_path=str(mpath), upload_path=dest_dir, overwrite=True 

530 ) 

531 except Exception as exc: # noqa: BLE001 

532 raise click.ClickException( 

533 f"Failed to upload teleport manifest: {exc}"

543def _landed(dataset_api, dest_dir: str, session_id: str, pushed_at: str) -> bool: 

544 """Whether the pod has acked THIS push's landing. 

545 

546 The ack must echo this manifest's ``pushed_at``: acks are never deleted, so 

547 a leftover from an earlier land of the same session id would otherwise read 

548 as an instant (false) landing on a re-push. 

549 """

551 return bool(ack) and ack.get("pushed_at") == pushed_at

554def _await_landing(dataset_api, dest_dir: str, session_id: str, pushed_at: str) -> bool: 

555 """Poll for the pod's ack of this push, for roughly fifteen seconds. 

556 

557 The pod's landing hook writes the sidecar right after it resumes (or opens) 

558 the session; seeing it means the manual landing kit is unnecessary. 

559 """ 

560 for attempt in range(_LANDING_POLLS): 

561 if attempt: 

562 time.sleep(_LANDING_POLL_SECONDS) 

563 if _landed(dataset_api, dest_dir, session_id, pushed_at): 

564 return True 

565 return False

568def _pod_holds( 

569 dataset_api, dest_dir: str, session_id: str, holder: str, project_id: int 

570) -> bool: 

571 """Whether a live pod has landed this session, so it may still be writing it. 

572 

573 Push names the pod in the baton before the pod has done anything, and the 

574 pod lands only once a browser Terminal tab has opened a shell in it, so a 

575 ``pod:`` holder plus a running terminal does not mean the pod has the 

576 session: a staged push nobody opened a tab for would wedge both push and 

577 pull behind ``--force`` with nothing to protect. The pod's ack for the 

578 CURRENT manifest is the evidence. A manifest that cannot be read keeps the 

579 gate closed rather than guessing. 

580 """ 

581 if not holder.startswith("pod:") or not _pod_alive(project_id): 

582 return False 

583 manifest = _read_remote_json(dataset_api, f"{dest_dir}/{session_id}.teleport.json") 

584 if not manifest: 

585 return True 

586 return _landed(dataset_api, dest_dir, session_id, manifest.get("pushed_at"))

677@session_group.command("push") 

678@click.argument("session_id", required=False)

708) 

709@click.pass_context 

710def push( 

711 ctx: click.Context, 

712 session_id: str | None, 

713 fork: bool, 

714 model: str | None, 

715 prompt: str | None, 

716 open_ui: bool, 

717 force: bool, 

718) -> None: 

719 """Push the current coding agent session onto a Hopsworks terminal pod. 

720 

721 Resolves the active session for this directory (or the ``SESSION_ID`` given 

722 as an argument), uploads its transcript into the project's HopsFS, starts 

723 the terminal pod (when the feature is enabled on the cluster), and opens the 

724 terminal in the browser. By default this is a baton hand-off: the pod 

725 becomes the canonical copy and the local transcript is renamed aside (unless 

726 it is the session you are currently in, which stays live locally). `--fork` 

727 keeps a live local copy instead. 

728 

729 Running it again re-stages the same session. It refuses only when the 

730 staged copy holds lines this machine does not have, until `hops session 

731 pull` brings them back or `--force` discards them. 

732 

733 Args: 

734 ctx: Click context. 

735 session_id: Explicit session id, or None for the active one. 

736 fork: Copy instead of hand off; leave the local session canonical. 

737 model: Model the pod resumes with. 

738 prompt: First instruction fed to the resumed session, or None. 

739 open_ui: Open the terminal in the browser after pushing. 

740 force: Push over a session a live pod still holds. 

741 """ 

742 slug = _cwd_slug() 

743 jsonl = _resolve_local_session(slug, session_id) 

744 resolved_id = _session_id_of(jsonl) 

745 already_away = jsonl.name.endswith(".away") 

746 

747 project = conn.get_project(ctx) 

748 dataset_api = project.get_dataset_api() 

749 dest_dir = f"{_teleport_root()}/{slug}" 

750 

751 # Ownership gate, mirroring pull: a live pod that has landed this session 

752 # holds the canonical copy and may carry appends never pulled. Re-pushing 

753 # the laptop's copy would overwrite the transport and, when the pod lands 

754 # it, `cp -f` the pod's live transcript too -- silently destroying that 

755 # work. A dead pod, a staged push the pod never landed, a laptop-held baton, 

756 # or a --fork push is not gated. 

757 if not fork and not force: 

758 holder = (_read_baton(dataset_api, dest_dir, resolved_id) or {}).get( 

759 "holder", "" 

760 ) 

761 if _pod_holds(dataset_api, dest_dir, resolved_id, holder, project.id): 

762 raise click.ClickException( 

763 f"A live pod holds this session ({holder}); it may have work you " 

764 f"have not pulled. `hops session stop && hops session pull` first " 

765 f"to bring it back, or `push --force` to overwrite the pod's copy " 

766 f"(its un-pulled work is lost)." 

767 ) 

768 

769 # Git sync runs first: it is the interactive part (consent, key prompt, 

770 # commit+push offer), so it must finish before any store mutation. A None 

771 # simply means no git context travels with this push. 

772 git_ctx = git_sync.maybe_collect(dataset_api, _teleport_root().rsplit("/", 1)[0]) 

773 

774 # mkdir is best-effort: an existing dir is fine, and a real problem will 

775 # surface loudly on the upload below. 

776 with contextlib.suppress(Exception): 

777 dataset_api.mkdir(dest_dir) 

778 # Upload a snapshot copy, not the live file: the session may be the one the 

779 # user is in, still being appended to. The baton's transferred_lines must 

780 # count exactly the bytes that were shipped — counting the live file after 

781 # the upload could include lines the pod never received, and pull would then 

782 # misread a plain divergence as "no common history" (common < baseline). 

783 # The staged copy is always replaced (a repeated push is a no-op), except 

784 # when it holds lines this snapshot does not: that is work the overwrite 

785 # would destroy, so it takes a pull or --force first. 

786 with tempfile.TemporaryDirectory() as tmp: 

787 snapshot = Path(tmp) / f"{resolved_id}.jsonl" 

788 shutil.copyfile(jsonl, snapshot) 

789 local_lines = snapshot.read_text(errors="ignore").splitlines() 

790 lines = len(local_lines) 

791 if not force: 

792 ahead = _staged_lines_ahead(dataset_api, dest_dir, resolved_id, local_lines) 

793 if ahead: 

794 raise click.ClickException( 

795 f"The staged copy of this session has {ahead} line(s) this " 

796 f"machine does not (work synced back from the pod, or a push " 

797 f"from another machine). `hops session pull` first to bring " 

798 f"it back, or `push --force` to overwrite it." 

799 ) 

800 try: 

801 dataset_api.upload( 

802 local_path=str(snapshot), upload_path=dest_dir, overwrite=True 

803 ) 

804 except Exception as exc: # noqa: BLE001 

805 raise click.ClickException(f"Failed to ship session: {exc}") from exc 

806 output.success("✓ Pushed session %s to %s", resolved_id, project.name) 

807 

808 ws_url = (_launch_pod(project) or {}).get("wsUrl") 

809 

810 # Manifest before baton: the manifest is the pod watcher's trigger and the 

811 # hard-error step (a dropped manifest strands the session: staged, never 

812 # landed). Writing the baton after it means a failed manifest upload never 

813 # leaves the store naming a pod holder for a session that pod will never 

814 # receive, which would make the next pull demand --force against nothing. 

815 # The pod's land path does not read the baton, so it need not precede the 

816 # trigger. 

817 manifest = _build_manifest( 

818 resolved_id, slug, "fork" if fork else "push", model, prompt, git=git_ctx 

819 ) 

820 _upload_manifest(dataset_api, dest_dir, resolved_id, manifest) 

821 

822 # Baton hand-off: record that the canonical copy now lives on the pod. Skip 

823 # for --fork (deliberate copy). 

824 host = socket.gethostname() 

825 if not fork and not _write_baton( 

826 dataset_api, 

827 dest_dir, 

828 resolved_id, 

829 holder=f"pod:{project.name}", 

830 prev_holder=f"laptop:{host}", 

831 lines=lines, 

832 ): 

833 output.warn( 

834 "Could not record the baton hand-off; the store does not name the " 

835 "pod as holder, so a pull from another machine will treat this as " 

836 "an unowned copy." 

837 ) 

838 

839 # Only now that the hand-off is fully staged, rename the local transcript 

840 # aside so it is not resumed by accident — except the session you are in 

841 # (its file is held open by a live `claude`), which stays live locally. A 

842 # transcript an earlier push already renamed aside stays where it is. 

843 if not fork and not already_away: 

844 if _is_active_session(jsonl): 

845 output.warn( 

846 "This looks like the session you are in, so it stays live on " 

847 "this machine. The baton points to %s; close it here, then " 

848 "`hops session pull` to reclaim it (once the pod has landed it, " 

849 "`hops session stop` first, or pull refuses).", 

850 project.name, 

851 ) 

852 else: 

853 away = jsonl.parent / (jsonl.name + ".away") 

854 with contextlib.suppress(Exception): 

855 jsonl.rename(away) 

856 (jsonl.parent / f"{resolved_id}.away.json").write_text( 

857 json.dumps( 

858 {"project": project.name, "host": host, "pushed_at": _now()} 

859 ) 

860 ) 

861 

862 pod_session_dir = f"~/.claude/projects/{slug}" 

863 pod_jsonl = f"/hopsfs/{_teleport_root()}/{slug}/{resolved_id}.jsonl" 

864 landing = [ 

865 f"mkdir -p {pod_session_dir}", 

866 f"cp {pod_jsonl} {pod_session_dir}/", 

867 f"claude --resume {resolved_id}", 

868 ] 

869 

870 terminal_url = _terminal_ui_url(project.id) 

871 # Open the browser only in interactive use: JSON mode is for machines, so it 

872 # just carries the URL. Best-effort — a headless box has no browser. 

873 opened = False 

874 if open_ui and not output.JSON_MODE: 

875 opened = _open_browser(terminal_url) 

876 

877 if output.JSON_MODE: 

878 # No poll in machine mode: one instant ack check, the caller re-lists 

879 # if it wants to wait. 

880 landed = _landed(dataset_api, dest_dir, resolved_id, manifest["pushed_at"]) 

881 output.print_json( 

882 { 

883 "session_id": resolved_id, 

884 "slug": slug, 

885 "project": project.name, 

886 "shipped_to": f"{dest_dir}/{resolved_id}.jsonl", 

887 "ws_url": ws_url, 

888 "terminal_url": terminal_url, 

889 "landed": landed, 

890 "landing_steps": landing, 

891 } 

892 ) 

893 return 

894 

895 _say_terminal(terminal_url, opened) 

896 # The landing kit only matters when the pod did NOT self-land, so wait 

897 # briefly for its ack and keep the happy path to the three ✓ lines. No pod 

898 # (feature disabled) means no ack will ever come: skip straight to the kit. 

899 if ws_url and _await_landing( 

900 dataset_api, dest_dir, resolved_id, manifest["pushed_at"] 

901 ): 

902 output.success( 

903 "✓ Landed on pod %s — open the Terminal tab to resume", project.name 

904 ) 

905 return 

906 if ws_url: 

907 output.info("Not landed yet — the pod lands it once its Terminal tab is open.") 

908 output.info("Or land it manually:") 

909 else: 

910 output.info("No terminal pod to land it; land it manually:") 

911 for step in landing: 

912 output.info(" %s", step)

1002@session_group.command("pull") 

1003@click.argument("session_id", required=False) 

1004@click.option(

1007 help="On divergence, keep the local transcript (the pod's copy is parked " 

1008 "aside, never lost).", 

1009)

1011 "--theirs", 

1012 is_flag=True, 

1013 help="On divergence, take the pod's transcript (your local copy is parked " 

1014 "aside, never lost).", 

1015) 

1016@click.option( 

1017 "--force", 

1018 is_flag=True, 

1019 help="Steal the baton from a live pod or another machine. Without it, " 

1020 "pull refuses to reclaim a session a live holder is still writing.", 

1021) 

1022@click.pass_context 

1023def pull( 

1024 ctx: click.Context, session_id: str | None, ours: bool, theirs: bool, force: bool 

1025) -> None: 

1026 """Pull a terminal session back onto this machine and take the baton. 

1027 

1028 Downloads the transcript from ``Users/<username>/teleport/<slug>/`` into 

1029 ``~/.claude/projects/<slug>/`` and prints the resume command. Given a 

1030 ``SESSION_ID`` the source slug is found across your staged sessions, so you 

1031 can pull a session staged from another directory from anywhere; resume still 

1032 needs a path that hashes to that slug, so the original cwd is printed to 

1033 ``cd`` into. Without an id, pulls this directory's single staged session. 

1034 

1035 Reclaiming is baton-aware. If the pod still holds a live terminal session, 

1036 pull refuses unless ``--force`` (you would be stealing from a process that 

1037 is still writing). If both sides advanced the transcript since the hand-off, 

1038 pull refuses until you pick ``--ours`` or ``--theirs``; the losing side is 

1039 parked to a sidecar, never destroyed. A successful pull flips the baton to 

1040 this machine as the very last step, so an interrupted pull leaves the pod 

1041 canonical rather than orphaning the claim here. 

1042 

1043 Args: 

1044 ctx: Click context. 

1045 session_id: Explicit session id (located project-wide), or None to pick 

1046 this directory's only staged one. 

1047 ours: On divergence, keep local. 

1048 theirs: On divergence, take the pod's copy. 

1049 force: Steal the baton from a live pod or another machine. 

1050 """ 

1051 if ours and theirs: 

1052 raise click.ClickException("--ours and --theirs are mutually exclusive.") 

1053 

1054 slug = _cwd_slug() 

1055 project = conn.get_project(ctx) 

1056 dataset_api = project.get_dataset_api() 

1057 

1058 # A positional id may name a session staged from another directory: find its 

1059 # slug project-wide so `pull <id>` works from anywhere. Without an id we pull 

1060 # this directory's single staged session. 

1061 origin_cwd: str | None = None 

1062 if session_id: 

1063 _validate_session_id(session_id) 

1064 located = _locate_session(dataset_api, session_id) 

1065 if not located: 

1066 raise click.ClickException( 

1067 f"No staged session {session_id!r} found. List yours with " 

1068 "`hops session list --all`." 

1069 ) 

1070 slug, origin_cwd = located 

1071 else: 

1072 staged = _remote_session_ids(dataset_api, slug) 

1073 if not staged: 

1074 raise click.ClickException( 

1075 "No sessions staged for this directory. Sessions are staged " 

1076 "per directory — run from the directory you pushed from, or " 

1077 "`hops session list --all`." 

1078 ) 

1079 if len(staged) > 1: 

1080 raise click.ClickException( 

1081 "Multiple sessions staged for this directory; pass a session id " 

1082 "to choose one: " + ", ".join(staged) 

1083 ) 

1084 session_id = staged[0] 

1085 

1086 dest_dir = f"{_teleport_root()}/{slug}" 

1087 remote_jsonl = f"{dest_dir}/{session_id}.jsonl" 

1088 local_dir = _CLAUDE_PROJECTS / slug 

1089 local_dir.mkdir(parents=True, exist_ok=True) 

1090 local_jsonl = local_dir / f"{session_id}.jsonl" 

1091 away = local_dir / f"{session_id}.jsonl.away" 

1092 

1093 baton = _read_baton(dataset_api, dest_dir, session_id) 

1094 host = socket.gethostname() 

1095 me = f"laptop:{host}" 

1096 

1097 # --- Ownership gate: --force is required only to take the baton from a 

1098 # holder that may still be writing (a live pod that has landed the session, 

1099 # or another laptop we have no liveness oracle for). A dead pod, a push the 

1100 # pod never landed, or a baton we already hold, is a frictionless reclaim. 

1101 pod_holds = False 

1102 if baton: 

1103 holder = baton.get("holder", "") 

1104 if holder.startswith("pod:"): 

1105 pod_holds = _pod_holds( 

1106 dataset_api, dest_dir, session_id, holder, project.id 

1107 ) 

1108 if pod_holds and not force: 

1109 raise click.ClickException( 

1110 f"A live pod has landed this session ({holder}) and may " 

1111 f"still be writing it. `hops session stop` first, or `pull " 

1112 f"--force` to take the baton anyway (the pod's later writes " 

1113 f"become orphans)." 

1114 ) 

1115 elif holder.startswith("laptop:") and holder != me and not force: 

1116 raise click.ClickException( 

1117 f"Another machine holds this session ({holder}). " 

1118 f"`pull --force` to steal the baton." 

1119 ) 

1120 

1121 # --- Download the remote transcript to a scratch file so we can compare 

1122 # before touching anything local. 

1123 with tempfile.TemporaryDirectory() as tmp: 

1124 scratch = Path(tmp) / f"{session_id}.jsonl" 

1125 try: 

1126 dataset_api.download(remote_jsonl, local_path=str(scratch), overwrite=True) 

1127 except Exception as exc: # noqa: BLE001 

1128 raise click.ClickException(f"Failed to pull session: {exc}") from exc 

1129 remote_text = scratch.read_text(errors="ignore") 

1130 remote_lines = remote_text.splitlines() 

1131 

1132 # The local candidate is the resumable copy if present, else the transcript 

1133 # push renamed aside on hand-off. 

1134 local_src = ( 

1135 local_jsonl if local_jsonl.is_file() else away if away.is_file() else None 

1136 ) 

1137 local_text = local_src.read_text(errors="ignore") if local_src else "" 

1138 local_lines = local_text.splitlines() if local_src else None 

1139 

1140 # --- Decide which content wins (axis: divergence). 

1141 parked: Path | None = None 

1142 take_remote: bool 

1143 if baton is None: 

1144 # Fork / legacy: a batonless copy. Overwrite only when told to, and 

1145 # never mint a baton (the --fork pusher opted out of ownership). 

1146 if local_lines is not None and local_lines != remote_lines and not theirs: 

1147 raise click.ClickException( 

1148 "A different local copy of this session exists. `pull --theirs` " 

1149 "to replace it (your copy is parked aside)." 

1150 ) 

1151 if theirs and local_src and local_lines != remote_lines: 

1152 parked = local_dir / f"{session_id}.jsonl.diverged-{_stamp()}" 

1153 take_remote = True 

1154 else: 

1155 baseline = baton.get("transferred_lines") or 0 

1156 if local_lines is None: 

1157 take_remote = True 

1158 else: 

1159 rel = _transcript_relation(local_lines, remote_lines, baseline) 

1160 if rel == "same": 

1161 take_remote = False 

1162 elif rel == "fast_forward": 

1163 take_remote = True 

1164 elif rel == "local_ahead": 

1165 take_remote = False 

1166 elif rel == "diverged": 

1167 if not (ours or theirs): 

1168 raise click.ClickException( 

1169 f"Local and pod transcripts diverged since the hand-off "

1612async def _pump(ws, warned: list[bool]) -> None: 

1613 """Stream server frames to stdout until the socket closes or the shell exits. 

1614 

1615 Output frames are written raw (they carry ANSI); ``size`` drives the wrap 

1616 warning; every other control frame (``clients``/``windows``/``pong``) is 

1617 ignored — a mirror only renders the byte stream. 

1618 """ 

1619 async for message in ws: 

1620 frame = ( 

1621 message if isinstance(message, str) else message.decode("utf-8", "replace") 

1622 ) 

1623 try: 

1624 msg = json.loads(frame) 

1625 except ValueError: 

1626 sys.stdout.buffer.write(frame.encode("utf-8", "replace")) 

1627 sys.stdout.buffer.flush() 

1628 continue 

1629 kind = msg.get("type") 

1630 if kind == "output": 

1631 sys.stdout.buffer.write(msg.get("data", "").encode("utf-8", "replace")) 

1632 sys.stdout.buffer.flush() 

1633 elif kind == "size": 

1634 _maybe_warn_size(msg, warned) 

1635 elif kind == "exit": 

1636 break

1639async def _drive(ws, mode: str) -> None: 

1640 """Run one attached mirror session: init, pump output, forward input. 

1641 

1642 Sends the init frame (size + role), then races the output pump against the 

1643 detach key. Whenever a raw tty is available the stdin reader is registered 

1644 so Ctrl-] detaches in both modes; in ``ro`` mode every other keystroke is 

1645 discarded, in ``rw`` mode the bytes are forwarded as input frames. The tty 

1646 is always restored on exit. 

1647 

1648 In ``rw`` the mirror also drives the shared window's geometry: the init 

1649 frame is ignored when attaching an existing session, so it pushes this 

1650 terminal's size explicitly on attach and again on every SIGWINCH, keeping 

1651 the view rendered at the local size instead of inheriting the driver's. 

1652 ``ro`` never resizes the session (the writer's terminal governs the grid). 

1653 """ 

1654 cols, rows = shutil.get_terminal_size((80, 24)) 

1655 await ws.send(json.dumps({"cols": cols, "rows": rows, "mode": mode})) 

1656 

1657 loop = asyncio.get_running_loop() 

1658 detached = asyncio.Event() 

1659 warned = [False] 

1660 fd = sys.stdin.fileno() 

1661 old_attrs = None 

1662 reader_added = False 

1663 

1664 has_tty = termios is not None and sys.stdin.isatty() 

1665 can_write = mode == "rw" and has_tty 

1666 

1667 def _read_stdin() -> None: 

1668 try: 

1669 data = os.read(fd, 4096) 

1670 except OSError: 

1671 return 

1672 if not data or _MIRROR_DETACH in data: 

1673 detached.set() 

1674 return 

1675 if not can_write: 

1676 return # read-only: only the detach key means anything 

1677 loop.create_task( 

1678 ws.send( 

1679 json.dumps({"type": "input", "data": data.decode("utf-8", "replace")}) 

1680 ) 

1681 ) 

1682 

1683 def _send_resize() -> None: 

1684 c, r = shutil.get_terminal_size((80, 24)) 

1685 loop.create_task(ws.send(json.dumps({"type": "resize", "cols": c, "rows": r}))) 

1686 

1687 # Raw mode even read-only, so the documented Ctrl-] detach actually fires 

1688 # without waiting for a newline. 

1689 if has_tty: 

1690 old_attrs = termios.tcgetattr(fd) 

1691 tty.setraw(fd) 

1692 loop.add_reader(fd, _read_stdin) 

1693 reader_added = True 

1694 

1695 winch_added = False 

1696 if can_write: 

1697 _send_resize() # apply our geometry now; the init frame was ignored 

1698 with contextlib.suppress(NotImplementedError, ValueError): 

1699 loop.add_signal_handler(signal.SIGWINCH, _send_resize) 

1700 winch_added = True 

1701 

1702 try: 

1703 recv_task = loop.create_task(_pump(ws, warned)) 

1704 detach_task = loop.create_task(detached.wait()) 

1705 done, pending = await asyncio.wait( 

1706 {recv_task, detach_task}, return_when=asyncio.FIRST_COMPLETED 

1707 ) 

1708 for t in pending: 

1709 t.cancel() 

1710 if detach_task in done: 

1711 raise _Detach 

1712 recv_task.result() # re-raise a ConnectionClosed so the caller reconnects 

1713 finally: 

1714 if winch_added: 

1715 with contextlib.suppress(NotImplementedError, ValueError): 

1716 loop.remove_signal_handler(signal.SIGWINCH) 

1717 if reader_added: 

1718 loop.remove_reader(fd) 

1719 if old_attrs is not None: 

1720 termios.tcsetattr(fd, termios.TCSADRAIN, old_attrs)

1723async def _mirror_session(project_id: int, write: bool) -> None: 

1724 """Connect to the project's terminal WS and mirror it, reconnecting on drop. 

1725 

1726 Mints a fresh proxy token per connect (they are short-lived and 

1727 non-renewable), so an expiry-driven close just reconnects with a new one. 

1728 A clean detach returns; a lost connection retries with backoff up to 

1729 :data:`_MIRROR_MAX_RETRIES` before giving up. 

1730 """ 

1731 # The asyncio client (websockets >= 13) is explicit about the connector and 

1732 # takes additional_headers; the top-level websockets.connect is the legacy 

1733 # asyncio impl with a different header kwarg, so import the new one directly. 

1734 # Lazy on purpose: websockets is the optional `terminal` extra, only mirror 

1735 # needs it. 

1736 try: 

1737 from websockets.asyncio.client import connect 

1738 from websockets.exceptions import ConnectionClosed 

1739 except ImportError as exc:
