1"""The health of a bronze, silver or gold analytics layer, for ``hops factory system status``. 

2 

3`collect` gathers the facts: the layer's job runs, and for each feature group 

4it built (silver and rejects, or each data mart's gold tables) its rows, when 

5it was last written against its freshness target, in silver its share of 

6rejected rows against the quality gate, and its file layout, read from the table's files under the terminal's HopsFS mount by 

7the hops-table-maintenance scanner. The page is the ML system status page 

8(`health.render`), which draws the tables section when the facts have one. 

9"""

45def _table_dir(project_name: str, name: str, version: int) -> Path: 

46 """Where the terminal mounts an offline feature group's table.""" 

47 root = Path(os.environ.get("HOPSFS_MOUNT", "/hopsfs")) 

48 return ( 

49 root 

50 / "featurestore" 

51 / f"{project_name.lower()}_featurestore.db" 

52 / f"{name}_{version}" 

53 )

56def _last_delta_commit(table: Path) -> datetime | None: 

57 """The time of the latest commit in a Delta table's log, or None.""" 

58 log = table / "_delta_log" 

59 commits = sorted(log.glob("*.json")) if log.is_dir() else [] 

60 for path in reversed(commits[-5:]): 

61 for line in path.read_text(encoding="utf-8").splitlines(): 

62 info = json.loads(line).get("commitInfo") if line.strip() else None 

63 if info and info.get("timestamp"): 

64 return datetime.fromtimestamp(info["timestamp"] / 1000, timezone.utc) 

65 return None

68def _layout(table: Path) -> dict | None: 

69 if not table.is_dir(): 

70 return None 

71 doctor = _scanner() 

72 fmt = doctor.detect_format(table) 

73 evidence = {"delta": doctor.analyze_delta, "iceberg": doctor.analyze_iceberg}.get( 

74 fmt, doctor.analyze_hudi 

75 )(table) 

76 return { 

77 "format": fmt, 

78 **(evidence.get("file_organization") or {}), 

79 "partitions": (evidence.get("partition_statistics") or {}).get( 

80 "partition_count" 

81 ), 

82 }

85def _rows(conn: Any, name: str, version: int) -> int | None: 

86 try: 

87 cur = conn.cursor() 

88 cur.execute(f'select count(*) from "{name}_{version}"') 

89 return int(cur.fetchone()[0]) 

90 except Exception: # noqa: BLE001 - a table Trino cannot read is reported without rows 

91 return None

94def _freshness(doc: dict, table: dict) -> float | None: 

95 """How stale a table may be: its data mart's target in gold, its cadence's job's in silver.""" 

96 if table.get("mart"): 

97 for mart in doc.get("marts") or []: 

98 if mart.get("slug") == table["mart"]: 

99 return mart.get("freshness_hours") 

100 return None 

101 target = (doc.get("freshness") or {}).get("max_age_hours") 

102 if isinstance(target, dict): 

103 target = target.get(table.get("cadence") or "") or max( 

104 target.values(), default=None 

105 ) 

106 return target

109def _table(project: Any, conn: Any, table: dict, doc: dict, now: datetime) -> dict: 

110 name, version, kind = table["name"], int(table.get("version", 1)), table["kind"] 

111 path = _table_dir(getattr(project, "name", ""), name, version) 

112 fact: dict[str, Any] = {"name": name, "version": version, "kind": kind} 

113 if table.get("mart"): 

114 fact["mart"] = table["mart"] 

115 if not path.is_dir(): 

116 fact["problems"] = ["missing: no table in the feature store"] 

117 return fact 

118 fact["rows"] = _rows(conn, name, version) if conn else None 

119 written = _last_delta_commit(path) 

120 fact["last_write"] = written.isoformat() if written else None 

121 fact["layout"] = _layout(path) 

122 problems = [] 

123 target = _freshness(doc, table) 

124 if kind in ("bronze", "silver", "gold") and written and target: 

125 age = (now - written) / timedelta(hours=1) 

126 fact["age_hours"] = round(age, 1) 

127 fact["max_age_hours"] = target 

128 if age > target: 

129 problems.append(f"stale: last written {age:.0f} h ago, target {target} h") 

130 layout = fact["layout"] or {} 

131 if ( 

132 layout.get("active_files", 0) >= SMALL_FILES_MIN 

133 and layout.get("small_file_ratio", 0) > SMALL_FILE_RATIO 

134 ): 

135 problems.append( 

136 f"small files: {layout['small_file_ratio']:.0%} of {layout['active_files']} files; compact with hops-table-maintenance" 

137 ) 

138 fact["problems"] = problems 

139 return fact

142def _reject_shares(tables: list[dict], doc: dict) -> None: 

143 """Flag each silver table whose rejects table holds more than the quality gate allows.""" 

144 limit = (doc.get("quality") or {}).get("max_reject_pct") 

145 rejects = {t["name"]: t for t in tables if t["kind"] == "rejects"} 

146 for table in tables: 

147 reject = rejects.get(f"{table['name']}_rejects") 

148 if table["kind"] != "silver" or not reject: 

149 continue 

150 good, bad = table.get("rows"), reject.get("rows") 

151 if good is None or bad is None or good + bad == 0: 

152 continue 

153 pct = round(100 * bad / (good + bad), 2) 

154 table["reject_pct"] = pct 

155 if limit is not None and pct > limit: 

156 table["problems"].append(f"rejects: {pct}% of rows, gate {limit}%")

159def collect(project: Any, doc: dict, slug: str, hours: int = 24) -> dict: 

160 """The facts of an analytics layer's status report. 

161 

162 Args: 

163 project: The project. 

164 doc: The layer's system.yaml. 

165 slug: The layer. 

166 hours: How far back to read the job runs. 

167 

168 Returns: 

169 The facts. 

170 """ 

171 now = datetime.now(timezone.utc) 

172 since = now - timedelta(hours=hours) 

173 from hopsworks.cli.commands.analytics import layer_jobs, layer_tables 

174 

175 jobs = [] 

176 for spec in layer_jobs(doc): 

177 try: 

178 jobs.append(health._job_runs(project, spec["name"], since)) 

179 except Exception as exc: # noqa: BLE001 

180 jobs.append({"name": spec["name"], "error": str(exc), "runs": []}) 

181 try: 

182 conn = project.get_trino_api().connect( 

183 catalog="delta", 

184 schema=f"{getattr(project, 'name', '').lower()}_featurestore", 

185 ) 

186 except Exception: # noqa: BLE001 - row counts are left out without Trino 

187 conn = None 

188 tables = [_table(project, conn, t, doc, now) for t in layer_tables(doc)] 

189 if conn: 

190 conn.close() 

191 _reject_shares(tables, doc) 

192 failed = sum( 

193 1 

194 for j in jobs 

195 for r in j["runs"] 

196 if r.get("final_status") in ("FAILED", "KILLED") 

197 ) 

198 # A job that is gone or cannot be listed has no runs to fail. 

199 broken = [j["name"] for j in jobs if j.get("missing") or j.get("error")] 

200 troubled = [t["name"] for t in tables if t.get("problems")] 

201 layer = doc.get("layer") or {} 

202 return { 

203 "system": { 

204 "slug": slug, 

205 "name": layer.get("name") or slug, 

206 "type": f"{layer.get('kind', 'silver')} layer", 

207 }, 

208 "generated": now.isoformat(timespec="seconds"), 

209 "hours": hours, 

210 "overall": "failing" 

211 if (failed or broken) and troubled 

212 else "degraded" 

213 if failed or broken or troubled 

214 else "healthy", 

215 "counts": { 

216 "jobs": len(jobs), 

217 "runs": sum(len(j["runs"]) for j in jobs), 

218 "failed_runs": failed, 

219 "broken_jobs": len(broken), 

220 "services": 0, 

221 "unhealthy_services": 0, 

222 "tables": len(tables), 

223 "table_problems": len(troubled), 

224 }, 

225 "jobs": jobs, 

226 "services": [], 

227 "tables": tables, 

228 }
