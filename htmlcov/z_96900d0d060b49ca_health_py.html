2 

3`collect` gathers the facts: every job the system created (from the same

32def _when(value: Any) -> datetime | None: 

33 if isinstance(value, (int, float)): 

34 return datetime.fromtimestamp(value / 1000, tz=timezone.utc) 

35 if isinstance(value, str) and value: 

36 try: 

37 parsed = datetime.fromisoformat(value.replace("Z", "+00:00")) 

38 except ValueError: 

39 return None 

40 return parsed if parsed.tzinfo else parsed.replace(tzinfo=timezone.utc) 

41 return value if isinstance(value, datetime) else None

44def _cpu_millis(value: str) -> float | None: 

45 if not value: 

46 return None 

47 if value.endswith("m"): 

48 return float(value[:-1]) 

49 if value.endswith("n"): 

50 return float(value[:-1]) / 1e6 

51 return float(value) * 1000

54def _memory_mib(value: str) -> float | None: 

55 if not value: 

56 return None 

57 for unit, factor in _UNITS.items(): 

58 if value.endswith(unit): 

59 return float(value[: -len(unit)]) * factor 

60 if value[-1:] in "KMG": 

61 return float(value[:-1]) * {"K": 1 / 1000, "M": 1, "G": 1000}[value[-1]] 

62 return float(value) / (1024 * 1024)

70 return "\n".join((text or "").strip().splitlines()[-lines:])

73def _job_runs(project: Any, name: str, since: datetime) -> dict: 

74 job = project.get_job_api().get_job(name) 

75 if job is None: 

76 return {"name": name, "missing": True, "runs": []} 

77 runs = [] 

78 for execution in job.get_executions() or []: 

79 submitted = _when(execution.submission_time) 

80 if submitted is None or submitted < since: 

81 continue 

82 run = { 

83 "id": execution.id, 

84 "state": execution.state, 

85 "final_status": execution.final_status, 

86 "submitted": submitted.isoformat(), 

87 "duration_s": round((execution.duration or 0) / 1000), 

88 } 

89 if execution.final_status in ("FAILED", "KILLED") or execution.state in ( 

90 "FAILED", 

91 "KILLED", 

92 "FRAMEWORK_FAILURE", 

93 "APP_MASTER_START_FAILED", 

94 "INITIALIZATION_FAILED", 

95 ): 

96 try: 

97 # A temporary directory, since the report runs in the system's 

98 # git work tree, where downloaded logs do not belong. 

99 with tempfile.TemporaryDirectory(prefix="hops-status-logs-") as tmp: 

100 stdout, stderr = execution.download_logs(path=tmp) 

101 run["log_tail"] = _tail(_read(stderr)) or _tail(_read(stdout)) 

102 except Exception as exc: # noqa: BLE001 - a missing log is reported, not fatal 

103 run["log_tail"] = f"(logs unavailable: {exc})" 

104 runs.append(run) 

105 schedule = getattr(job, "job_schedule", None) 

106 return { 

107 "name": name, 

108 "type": getattr(job, "job_type", None), 

109 "scheduled": bool(schedule and getattr(schedule, "enabled", False)), 

110 "runs": sorted(runs, key=lambda r: r["submitted"], reverse=True), 

111 }

119 )

123def _pods(prefixes: dict[str, str]) -> dict[str, list[dict]]: 

124 """The pods of each named component, by pod-name prefix, with their resource use.""" 

125 listed = _kubectl("get", "pods", "-o", "json") 

126 if not listed: 

127 return {} 

128 usage = {} 

129 for line in _kubectl("top", "pods", "--no-headers", "--containers").splitlines(): 

130 parts = line.split() 

131 if len(parts) >= 4: 

132 pod = usage.setdefault(parts[0], {"cpu_m": 0.0, "memory_mib": 0.0}) 

133 pod["cpu_m"] += _cpu_millis(parts[2]) or 0 

134 pod["memory_mib"] += _memory_mib(parts[3]) or 0 

135 found: dict[str, list[dict]] = {name: [] for name in prefixes} 

136 for pod in json.loads(listed).get("items", []): 

137 name = pod["metadata"]["name"] 

138 owner = next((c for c, p in prefixes.items() if name.startswith(p)), None) 

139 if owner is None: 

140 continue 

141 statuses = pod.get("status", {}).get("containerStatuses", []) 

142 limits_cpu = limits_mem = 0.0 

143 for container in pod["spec"].get("containers", []): 

144 limits = container.get("resources", {}).get("limits", {}) 

145 limits_cpu += _cpu_millis(limits.get("cpu", "")) or 0 

146 limits_mem += _memory_mib(limits.get("memory", "")) or 0 

147 terminated = [ 

148 { 

149 "container": s["name"], 

150 "reason": s["lastState"]["terminated"].get("reason"), 

151 "exit_code": s["lastState"]["terminated"].get("exitCode"), 

152 } 

153 for s in statuses 

154 if s.get("lastState", {}).get("terminated") 

155 ] 

156 found[owner].append( 

157 { 

158 "pod": name, 

159 "phase": pod.get("status", {}).get("phase"), 

160 "ready": f"{sum(1 for s in statuses if s.get('ready'))}/{len(statuses)}", 

161 "restarts": sum(s.get("restartCount", 0) for s in statuses), 

162 "last_terminations": terminated, 

163 "cpu_m": usage.get(name, {}).get("cpu_m"), 

164 "memory_mib": usage.get(name, {}).get("memory_mib"), 

165 "cpu_limit_m": limits_cpu or None, 

166 "memory_limit_mib": limits_mem or None, 

167 } 

168 ) 

169 return found

172def collect(project: Any, doc: dict, slug: str, hours: int = 24) -> dict: 

173 """The facts of the report: jobs over the last `hours`, what the feature pipelines wrote, deployments and apps with their pods. 

174 

175 Args: 

176 project: The project. 

177 doc: The system's system.yaml. 

178 slug: The system. 

179 hours: How far back to read the job runs. 

180 

181 Returns: 

182 The facts. 

183 """ 

184 since = datetime.now(timezone.utc) - timedelta(hours=hours) 

185 assets = teardown.inventory(doc, slug) 

186 job_names = [a.name for a in assets if a.kind == "job" and not a.name.endswith("*")] 

187 jobs = [] 

188 for name in job_names: 

189 try: 

190 jobs.append(_job_runs(project, name, since)) 

191 except Exception as exc: # noqa: BLE001 

192 jobs.append({"name": name, "error": str(exc), "runs": []}) 

193 

194 services = [] 

195 serving = project.get_model_serving() 

196 for asset in assets: 

197 if asset.kind == "deployment": 

198 deployment = serving.get_deployment(asset.name) 

199 state = None 

200 if deployment is not None: 

201 try: 

202 state = deployment.get_state().status 

203 except Exception as exc: # noqa: BLE001 

204 state = f"unknown ({exc})" 

205 services.append( 

206 { 

207 "kind": "deployment", 

208 "name": asset.name, 

209 "state": state or "missing", 

210 "prefix": f"{asset.name}-predictor", 

211 } 

212 ) 

213 elif asset.kind == "app": 

214 app = project.get_app_api().get_app(asset.name) 

215 services.append( 

216 { 

217 "kind": "app", 

218 "name": asset.name, 

219 "state": getattr(app, "state", None) or "missing", 

220 "url": getattr(app, "app_url", None), 

221 "prefix": f"pythonapp-{getattr(project, 'name', '')}--{asset.name}", 

222 } 

223 ) 

224 pods = _pods({s["name"]: s.pop("prefix") for s in services}) 

225 for service in services: 

226 service["pods"] = pods.get(service["name"], []) 

227 # Imported here: pipeline_data reads tables through silver_status, which imports this module. 

228 from hopsworks.cli import pipeline_data 

229 

230 pipelines = pipeline_data.collect(project, doc, hours) 

231 troubled = [p["name"] for p in pipelines if p["problems"]] 

232 

233 failed = sum( 

234 1 

235 for j in jobs 

236 for r in j["runs"] 

237 if r.get("final_status") in ("FAILED", "KILLED") 

238 ) 

239 # A job that is gone or cannot be listed has no runs to fail. 

240 broken = [j["name"] for j in jobs if j.get("missing") or j.get("error")] 

241 unhealthy = [ 

242 s["name"] 

243 for s in services 

244 if str(s["state"]).upper() not in ("RUNNING", "IDLE") 

245 or any( 

246 p["restarts"] 

247 or p["phase"] != "Running" 

248 or p["ready"].split("/")[0] != p["ready"].split("/")[1] 

249 for p in s["pods"] 

250 ) 

251 ] 

252 return { 

253 "system": { 

254 "slug": slug, 

255 "name": (doc.get("system") or {}).get("name") or slug, 

256 "type": (doc.get("requirements") or {}).get("system_type"), 

257 }, 

258 "generated": datetime.now(timezone.utc).isoformat(timespec="seconds"), 

259 "hours": hours, 

260 "overall": "failing" 

261 if (failed or broken) and (unhealthy or troubled) 

262 else "degraded" 

263 if failed or broken or unhealthy or troubled 

264 else "healthy", 

265 "counts": { 

266 "jobs": len(jobs), 

267 "runs": sum(len(j["runs"]) for j in jobs), 

268 "failed_runs": failed, 

269 "broken_jobs": len(broken), 

270 "services": len(services), 

271 "unhealthy_services": len(unhealthy), 

272 "pipelines": len(pipelines), 

273 "pipeline_problems": len(troubled), 

274 }, 

275 "jobs": jobs, 

276 "services": services, 

277 "pipelines": pipelines, 

278 }

281def summarize(facts: dict, timeout: int = 180) -> str | None: 

282 """Claude's short account of what failed and why, as HTML paragraphs and lists, or None. 

283 

284 Args: 

285 facts: The facts of the report. 

286 timeout: Seconds to wait for Claude. 

287 

288 Returns: 

289 The summary, None when Claude is not there or did not answer. 

290 """ 

291 if not shutil.which("claude"): 

292 return None 

293 prompt = ( 

294 "You are writing the summary section of a health report for an ML system or an analytics layer on Hopsworks. " 

295 "From the JSON facts below, write at most 150 words as an HTML fragment using only <p>, <ul>, " 

296 "<li>, <strong> and <code>: first one sentence on the overall health, then, for each failed " 

297 "job run, unhealthy deployment or app, or table with problems (stale, rejects over the gate, small files), " 

298 "what failed and the likely cause from its log tail " 

299 "or pod state (OOMKilled means raise the memory), and what to do. Say plainly when everything " 

300 "is healthy. No headings, no markdown, no preamble.\n\n" 

301 + json.dumps(facts, default=str) 

302 ) 

303 try: 

304 done = subprocess.run( 

305 ["claude", "-p", "--model", "haiku", prompt], 

306 capture_output=True, 

307 text=True, 

308 timeout=timeout, 

309 check=False, 

310 ) 

311 except subprocess.TimeoutExpired: 

312 return None 

313 text = done.stdout.strip() if done.returncode == 0 else "" 

314 # Only the tags asked for survive; anything else is shown as text. 

315 text = re.sub(r"```(?:html)?", "", text).strip() 

316 allowed = re.compile(r"</?(p|ul|li|strong|code)>") 

317 parts = re.split(r"(</?[a-zA-Z][^>]*>)", text) 

318 return ( 

319 "".join( 

320 p if allowed.fullmatch(p) else html.escape(p) if p.startswith("<") else p 

321 for p in parts 

322 ) 

323 or None 

324 )
